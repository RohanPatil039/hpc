#include <mpi.h>
#include <iostream>
#include <vector>
#include <algorithm>

int main(int argc, char* argv[]) {
    MPI_Init(&argc, &argv);
    int rank, size;
    MPI_Comm_rank(MPI_COMM_WORLD, &rank);
    MPI_Comm_size(MPI_COMM_WORLD, &size);

    int N;
    std::vector<int> full;

    if (rank == 0) {
        std::cout << "Enter number of elements (must be divisible by " << size << "): ";
        std::cin >> N;

        if (N % size != 0) {
            std::cerr << "Error: Number of elements must be divisible by number of processes.\n";
            MPI_Abort(MPI_COMM_WORLD, 1);
        }

        full.resize(N);
        std::cout << "Enter " << N << " integers:\n";
        for (int i = 0; i < N; ++i) {
            std::cin >> full[i];
        }
    }

    // Broadcast N to all processes
    MPI_Bcast(&N, 1, MPI_INT, 0, MPI_COMM_WORLD);
    int chunk = N / size;
    std::vector<int> local(chunk);

    // Scatter the input array
    MPI_Scatter(full.data(), chunk, MPI_INT, local.data(), chunk, MPI_INT, 0, MPI_COMM_WORLD);
    std::sort(local.begin(), local.end());

    // Odd-even transposition sort
    for (int phase = 0; phase < size; ++phase) {
        int partner = (phase % 2 == 0)
            ? (rank % 2 == 0 ? rank + 1 : rank - 1)
            : (rank % 2 == 0 ? rank - 1 : rank + 1);
        if (partner < 0 || partner >= size) continue;

        std::vector<int> temp(chunk);
        MPI_Sendrecv(local.data(), chunk, MPI_INT, partner, 0,
            temp.data(), chunk, MPI_INT, partner, 0,
            MPI_COMM_WORLD, MPI_STATUS_IGNORE);

        std::vector<int> merged(chunk * 2);
        std::merge(local.begin(), local.end(), temp.begin(), temp.end(), merged.begin());

        if (rank < partner)
            std::copy(merged.begin(), merged.begin() + chunk, local.begin());
        else
            std::copy(merged.end() - chunk, merged.end(), local.begin());
    }

    // Gather the sorted chunks
    if (rank == 0) full.resize(N); // Resize to receive final sorted array
    MPI_Gather(local.data(), chunk, MPI_INT, full.data(), chunk, MPI_INT, 0, MPI_COMM_WORLD);

    if (rank == 0) {
        std::cout << "Sorted: ";
        for (int x : full) std::cout << x << " ";
        std::cout << "\n";
    }

    MPI_Finalize();
    return 0;
}
